The terminal can only follow your shell if the shell reports its folder (OSC 7). macOS does that for Apple's Terminal only, so it has to be set up once. Peach Commander will not edit your shell's startup file: add these lines to ~/.zshrc yourself.
# Report the working directory to the terminal (OSC 7), for Peach Commander
autoload -Uz add-zsh-hook
_pc_osc7() { printf '\033]7;file://%s%s\007' "$HOST" "${PWD// /%20}" }
add-zsh-hook precmd _pc_osc7
